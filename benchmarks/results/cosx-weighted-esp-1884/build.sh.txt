#!/bin/bash
set -euo pipefail
cd /data/jzzeng/qc-dft-cosx-esp-crossover-1884
export PATH=/data/jzzeng/qc-provider-1886-tools/cmake/bin:/data/jzzeng/qc-provider-1886-tools/cuda-12.9/bin:/data/jzzeng/qc-provider-1886-tools/bin:/data/jzzeng/qc-provider-1886-tools/venv-rhf/bin:$PATH
export CCACHE_DIR=/data/jzzeng/ccache CCACHE_BASEDIR=$PWD TMPDIR=/data/jzzeng/qc-provider-1886-tools/tmp
export PYTHONPATH=python:.:tests/python PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
out=$PWD/.artifacts/1884-cosx-esp-crossover
sha256sum -c "$out/source-sha256.txt" > "$out/build-source-verified.txt"
ccache --version > "$out/cache-version.txt"
ccache --show-stats > "$out/cache-before.txt"
trap 'ccache --show-stats > "$out/cache-after.txt"' EXIT
cmake --preset cuda-release-sm120 -DCMAKE_CXX_COMPILER_LAUNCHER=ccache -DCMAKE_CUDA_COMPILER_LAUNCHER=ccache -DGENERATIVEQC_ENABLE_STATIONARY_FORCE_AOT=OFF -DGENERATIVEQC_ENABLE_STATIONARY_CPU_FORCE_AOT=OFF
ninja -C build/cuda-release-sm120 -t commands generativeqc_cosx_cuda_tests > "$out/commands.txt"
python - <<'PY'
from pathlib import Path
commands = Path('.artifacts/1884-cosx-esp-crossover/commands.txt').read_text().splitlines()
compiles = [command for command in commands if ' -c ' in command and ('.cu' in command or '.cpp' in command)]
assert compiles and all('ccache' in command for command in compiles)
PY
ninja -C build/cuda-release-sm120 -j 4 generativeqc_cosx_cuda_tests
