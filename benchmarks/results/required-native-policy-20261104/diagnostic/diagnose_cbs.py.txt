"""Untimed one-endpoint diagnosis; never a substitute qualification sample."""

import json
import os
import sys
import time
from pathlib import Path

from generativeqc import CBSBasis, CCSDTCBSCalculator, Calculator, load_basis


def event(kind, **details):
    """Flush group boundaries even when a later native operation never returns."""
    print(json.dumps({"event": kind, "monotonic_seconds": time.monotonic(), **details}), flush=True)


original = Calculator.singlepoint_components


def traced_components(calculator, *arguments, **keywords):
    """Observe existing execution; preserve every input, output and exception."""
    started = time.monotonic()
    identity = {"method": calculator._method, "basis_identity": calculator._basis.identity,
                "companions": keywords.get("companions"), "properties": keywords.get("properties")}
    event("group_started", **identity)
    try:
        results = original(calculator, *arguments, **keywords)
    except BaseException as error:
        event("group_failed", **identity, error_type=type(error).__name__, error=str(error))
        raise
    event("group_completed", **identity, seconds=time.monotonic() - started,
          results={method: {"energy": result.energy, "iterations": result.iterations,
                            "backend": result.executed_backend} for method, result in results.items()})
    return results


Calculator.singlepoint_components = traced_components
reference_root = Path.cwd() / "tests/reference_data/cbs"
reference = json.loads((reference_root / "h2-dz-tz.json").read_text())
bases = tuple(CBSBasis(load_basis(reference_root / reference[field]), cardinal)
              for field, cardinal in (("low_basis", 2), ("high_basis", 3)))
event("diagnostic_started", qualification=False, python_source=sys.path[:2], workload=reference["case"])
owner = CCSDTCBSCalculator(bases, policy="standard", device="cpu")
result = owner.singlepoint(reference["atoms"], properties=("energy", "forces"))
event("diagnostic_completed", qualification=False, energy=result.energy, forces=result.forces.tolist(),
      convergence=result.convergence, execution_groups=result.execution_groups,
      components=[{"roles": component.roles, "method": component.method,
                   "basis_identity": component.basis_identity, "coefficient": component.coefficient,
                   "energy": component.result.energy, "forces": component.result.forces.tolist()}
                  for component in result.components])
if path := os.environ.get("GENERATIVEQC_DIAGNOSTIC_PACKAGED"):
    import ctypes

    observer = ctypes.CDLL(path)
    query = observer.generativeqc_diagnostic_packaged_counts
    query.argtypes = [ctypes.POINTER(ctypes.c_uint64), ctypes.POINTER(ctypes.c_uint64)]
    query.restype = None
    shells, primitives = ctypes.c_uint64(), ctypes.c_uint64()
    query(ctypes.byref(shells), ctypes.byref(primitives))
    event("diagnostic_native_work", qualification=False, shell_calls=shells.value,
          primitive_evaluations=primitives.value)
