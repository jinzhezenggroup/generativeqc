"""Frozen 1.05 ABBA/BAAB complete-endpoint qualification for policy retirement."""

import ctypes
import hashlib
import json
import os
import statistics
import subprocess
import sys
from pathlib import Path

from tools.generate_build_identity import _inventory, _source_identity

root = Path.cwd()
output = root / ".artifacts/required-native-policy-20261104"
patch = subprocess.check_output(["git", "diff", "--binary", "HEAD"])
assert not patch, "Measure a committed candidate"
library_path = root / "build/native-runtime-policy/libgenerativeqc.so"
library = ctypes.CDLL(str(library_path))
library.generativeqc_get_source_identity.restype = ctypes.c_char_p
actual_identity = library.generativeqc_get_source_identity().decode()
expected_identity = _source_identity(root, _inventory(root, root / "cmake/GenerativeQCSourceIdentity.json"))
assert actual_identity == expected_identity, "Selected library does not match full candidate inputs"
baseline = subprocess.check_output(["git", "rev-parse", "1622554d9"], text=True).strip()
subprocess.run(["git", "diff", "--exit-code", baseline, "HEAD", "--", "src", "include", "cmake",
                "python/generativeqc_compiler", "CMakeLists.txt"], check=True)
affinity = max(os.sched_getaffinity(0))
os.sched_setaffinity(0, {affinity})
environment = dict(os.environ, GENERATIVEQC_LIBRARY=str(library_path), GENERATIVEQC_PROFILE="off",
                   OPENBLAS_NUM_THREADS="1", OMP_NUM_THREADS="1")
report = dict(protocol_revision=1, baseline=baseline,
              candidate=subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
              candidate_patch_sha256=hashlib.sha256(patch).hexdigest(),
              native_library_sha256=hashlib.sha256(library_path.read_bytes()).hexdigest(),
              actual_native_source_identity=actual_identity, expected_native_source_identity=expected_identity,
              affinity_cpu=affinity, ratio_gate=1.05, samples_per_block=21, warmups_per_block=3,
              source_gate="Actual native source query equals the full current input inventory before timing",
              results=[])
for workload in ("precision-rhf", "precision-pbe0", "cbs-standard"):
    for order in ("ABBA", "BAAB"):
        records = []
        for index, variant in enumerate(order):
            source = root if variant == "B" else output / "baseline"
            environment["PYTHONPATH"] = str(source / "python")
            completed = subprocess.run([sys.executable, str(output / "probe.py"), workload],
                                       env=environment, capture_output=True, text=True, timeout=600)
            if completed.returncode:
                (output / f"{workload}-{order}-{index}-{variant}-failure.txt").write_text(completed.stdout + completed.stderr)
                raise RuntimeError("Endpoint probe failed; retain its raw diagnostic")
            record = json.loads(completed.stdout)
            (output / f"{workload}-{order}-{index}-{variant}.json").write_text(json.dumps(record) + "\n")
            records.append((variant, record))
        publications = [sample for _, record in records
                        for sample in (record["first_call"], *record["warmups"], *record["samples"])]
        reference = publications[0]
        science_work = all(abs(sample["energy"] - reference["energy"]) <= 1e-12
                           and all(abs(value - expected) <= 1e-12
                                   for row, target in zip(sample["forces"], reference["forces"], strict=True)
                                   for value, expected in zip(row, target, strict=True))
                           and sample["semantic_work"] == reference["semantic_work"]
                           and sample["executed_backend"] == reference["executed_backend"] == "cpu_reference"
                           for sample in publications)
        medians = {variant: statistics.median(sample["seconds"] for chosen, record in records if chosen == variant
                                            for sample in record["samples"]) for variant in ("A", "B")}
        ratio = medians["B"] / medians["A"]
        report["results"].append(dict(workload=workload, order=order, medians=medians, ratio=ratio,
                                      science_work_passed=science_work, performance_passed=ratio <= report["ratio_gate"]))
        print(json.dumps(report["results"][-1]), flush=True)
report["passed"] = all(row["science_work_passed"] and row["performance_passed"] for row in report["results"])
(output / "summary.json").write_text(json.dumps(report, indent=2) + "\n")
raise SystemExit(0 if report["passed"] else 1)
