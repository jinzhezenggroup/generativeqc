"""One-off complete CPU precision/CBS endpoints with explicit semantic work."""

import json
import sys
import time
from dataclasses import asdict
from pathlib import Path

from generativeqc import CBSBasis, CCSDTCBSCalculator, GridSpec, KsOptions, PrecisionCalculator, load_basis

workload = sys.argv[1]
reference_root = Path.cwd() / "tests/reference_data/cbs"
reference = json.loads((reference_root / "h2-dz-tz.json").read_text())
atoms = tuple((atom[0], tuple(atom[1])) for atom in reference["atoms"])
if workload == "cbs-standard":
    bases = tuple(CBSBasis(load_basis(reference_root / reference[field]), cardinal)
                  for field, cardinal in (("low_basis", 2), ("high_basis", 3)))
else:
    options = dict(method="rhf" if workload == "precision-rhf" else "pbe0-rks",
                   policy="fast", device="cpu", basis="sto-3g")
    if workload == "precision-pbe0":
        options["ks_options"] = KsOptions(grid=GridSpec.treutler(0))

def correlation_value(component, name):
    """Unavailable per-component diagnostics remain unknown, never measured zero."""
    return None if component.result.correlation is None else getattr(component.result.correlation, name)


samples = []
for index in range(25):
    started = time.perf_counter()
    owner = CCSDTCBSCalculator(bases, policy="standard", device="cpu") if workload == "cbs-standard" else PrecisionCalculator(**options)
    result = owner.singlepoint(atoms, properties=("energy", "forces"))
    if workload != "cbs-standard":
        owner.clear_cache()
    elapsed = time.perf_counter() - started
    assert result.converged and result.forces is not None
    if workload == "cbs-standard":
        raw = {
            "convergence": result.convergence,
            "execution_groups": result.execution_groups,
            "components": [
                {"roles": component.roles, "method": component.method,
                 "basis_identity": component.basis_identity, "coefficient": component.coefficient,
                 "seconds": component.elapsed_seconds, "iterations": component.result.iterations,
                 "energy": component.result.energy, "forces": component.result.forces.tolist(),
                 "ccsd_iterations": correlation_value(component, "ccsd_iterations"),
                 "response_iterations": correlation_value(component, "response_iterations"),
                 "response_restarts": correlation_value(component, "response_restarts"),
                 "planned_endpoint_peak_bytes": correlation_value(component, "planned_endpoint_peak_bytes"),
                 "numeric_capacity_bytes": correlation_value(component, "numeric_capacity_bytes")}
                for component in result.components
            ],
        }
        work = {
            "execution_groups": [{name: value for name, value in group.items() if name != "elapsed_seconds"}
                                 for group in result.execution_groups],
            "components": [{name: value for name, value in component.items() if name not in ("seconds", "energy", "forces")}
                           for component in raw["components"]],
            "convergence": result.convergence,
            "opaque_scf_fock_counts": None,
        }
    else:
        raw = result.precision_validation.to_dict()
        work = {
            "status": result.precision_validation.status,
            "model_identity": result.precision_validation.model_identity,
            "solve_count": result.precision_validation.solve_count,
            "solves": [{name: value for name, value in asdict(solve).items() if name != "seconds"}
                       for solve in result.precision_validation.solves],
        }
    samples.append(dict(seconds=elapsed, energy=result.energy, forces=result.forces.tolist(),
                        executed_backend=result.executed_backend, semantic_work=work, raw=raw))

print(json.dumps(dict(workload=workload, python_source=sys.path[:2],
                     endpoint="Construction plus synchronous full E/F publication and transient-owner cleanup",
                     input_setup="Immutable atoms, basis snapshots and KS options prepared before each block",
                     first_call=samples[0], warmups=samples[1:4], samples=samples[4:]), allow_nan=False))
