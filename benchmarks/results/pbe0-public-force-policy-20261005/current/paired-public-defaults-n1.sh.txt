#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-current-defaults-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/endpoint-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
"$python" .artifacts/check-source.py > "$output/source-check.txt"
sha256sum .artifacts/public-default-endpoint.py .artifacts/verify-public-default.py .artifacts/paired-public-defaults-n1.sh .artifacts/gpu-env-n1.sh > "$output/driver.sha256"
for atoms in 48 96; do
  "$python" -m benchmarks.readme_pbe0 reference --atoms "$atoms" --basis-file benchmarks/results/pbe0-def2-svp-20261003/def2-svp-ho.json --repeats 5 --output "$output/reference-$atoms.json" > "$output/reference-$atoms.log" 2>&1
  arms=(dense-control default)
  if [[ "$atoms" == 96 ]]; then arms=(default dense-control); fi
  for arm in "${arms[@]}"; do
    export GENERATIVEQC_STATIONARY_CACHE="$output/cache-$arm-$atoms"
    mkdir -p "$GENERATIVEQC_STATIONARY_CACHE"
    ccache --show-stats > "$output/ccache-before-$arm-$atoms.txt"
    "$python" .artifacts/public-default-endpoint.py --force-policy "$arm" native --atoms "$atoms" --basis-file benchmarks/results/pbe0-def2-svp-20261003/def2-svp-ho.json --repeats 5 --reference "$output/reference-$atoms.json" --output "$output/$arm-$atoms.json" > "$output/$arm-$atoms.log" 2>&1
    "$python" .artifacts/verify-public-default.py "$output/$arm-$atoms.json" "$output/reference-$atoms.json" > "$output/verified-$arm-$atoms.json"
    ccache --show-stats > "$output/ccache-after-$arm-$atoms.txt"
  done
done
