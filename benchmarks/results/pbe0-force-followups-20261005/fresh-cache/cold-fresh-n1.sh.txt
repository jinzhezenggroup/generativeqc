#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-current-defaults-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/cold-fresh-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
"$python" .artifacts/check-source.py > "$output/source-check.txt"
sha256sum .artifacts/cold-fresh-n1.sh .artifacts/verify-cold-repeat.py .artifacts/public-default-endpoint.py > "$output/driver.sha256"
"$python" -m benchmarks.readme_pbe0 reference --atoms 96 --basis-file benchmarks/results/pbe0-def2-svp-20261003/def2-svp-ho.json --repeats 1 --output "$output/reference-96.json" > "$output/reference.log" 2>&1
# Separate processes, cold native SCF owners and separate initially empty
# generated-artifact caches. No arm inherits setup paid by an earlier arm.
index=0
for arm in dense-control default default dense-control; do
  index=$((index + 1))
  export GENERATIVEQC_STATIONARY_CACHE="$output/cache-$index-$arm"
  mkdir -p "$GENERATIVEQC_STATIONARY_CACHE"
  "$python" .artifacts/public-default-endpoint.py --force-policy "$arm" native --atoms 96 --basis-file benchmarks/results/pbe0-def2-svp-20261003/def2-svp-ho.json --repeats 1 --reference "$output/reference-96.json" --output "$output/$index-$arm.json" > "$output/$index-$arm.log" 2>&1
  "$python" .artifacts/verify-cold-repeat.py "$output/$index-$arm.json" "$output/reference-96.json" > "$output/verified-$index-$arm.json"
done
