// One finite engineering process: no geometry, solver call, matrix or SCF.
#include <cuda_runtime_api.h>
#include <cublas_v2.h>
#include <cusolverDn.h>
#include <dlfcn.h>
#include <iostream>
#include <stdexcept>
#include <string>

namespace {
std::size_t sample_after = 0;
cudaError_t observed_mem_info(std::size_t* available, std::size_t* total) {
  auto status = cudaMemGetInfo(available, total);
  if (status == cudaSuccess) sample_after = *available;
  return status;
}
int cleanup_failures = 0;
struct Resources {
  cudaStream_t stream = nullptr;
  cublasHandle_t blas = nullptr;
  cusolverDnHandle_t solver = nullptr;
  cusolverDnParams_t parameters = nullptr;
  ~Resources() {
    auto record = [](const char* name, int status) {
      std::cout << "cleanup " << name << '=' << status << '\n';
      if (status) ++cleanup_failures;
    };
    if (parameters) record("parameters", cusolverDnDestroyParams(parameters));
    if (solver) record("solver", cusolverDnDestroy(solver));
    if (blas) record("blas", cublasDestroy(blas));
    if (stream) record("stream", cudaStreamDestroy(stream));
  }
};
void vendor_check(int status) {
  if (status) throw std::runtime_error("vendor initialization failed: status=" + std::to_string(status));
}
void library(const char* name, void* symbol) {
  Dl_info info{};
  if (!dladdr(symbol, &info) || !info.dli_fname) throw std::runtime_error("loaded DSO unavailable");
  std::cout << "loaded " << name << '=' << info.dli_fname << '\n';
}
}

// Observe the real successful runtime snapshot consumed by the production
// guard; synchronization, runtime status and every numeric value stay real.
#define cudaMemGetInfo observed_mem_info
#include "scf/cuda/reference_export.cuh"
#undef cudaMemGetInfo

int main() {
  namespace detail = generativeqc::scf::reference_detail;
  try {
    {
      Resources resources;
      generativeqc_tensor::cuda_check(cudaStreamCreate(&resources.stream));
      int runtime = 0, driver = 0;
      generativeqc_tensor::cuda_check(cudaRuntimeGetVersion(&runtime));
      generativeqc_tensor::cuda_check(cudaDriverGetVersion(&driver));
      std::cout << "runtime_version=" << runtime << " driver_api=" << driver << '\n';
      library("cudart", reinterpret_cast<void*>(&cudaMemGetInfo));
      library("cublas", reinterpret_cast<void*>(&cublasCreate_v2));
      library("cusolver", reinterpret_cast<void*>(&cusolverDnCreate));
      auto before = detail::free_bytes(resources.stream);
      vendor_check(cublasCreate(&resources.blas));
      vendor_check(cublasSetStream(resources.blas, resources.stream));
      vendor_check(cublasSetPointerMode(resources.blas, CUBLAS_POINTER_MODE_HOST));
      vendor_check(cusolverDnCreate(&resources.solver));
      vendor_check(cusolverDnSetStream(resources.solver, resources.stream));
      vendor_check(cusolverDnCreateParams(&resources.parameters));
      constexpr std::size_t allowance = 192ULL << 20;
      bool refused = false;
      std::size_t returned = 0;
      std::string message;
      try {
        returned = detail::retained_bytes(resources.stream, before, allowance);
      } catch (const std::length_error& error) {
        refused = true;
        message = error.what();
      }
      auto expected = before > sample_after ? before - sample_after : 0;
      std::cout << "free_before_bytes=" << before << " free_after_bytes=" << sample_after
                << " observed_delta_bytes=" << expected << " allowance_bytes=" << allowance << '\n';
      std::cout << "provider_refused=" << refused << '\n';
      if (refused) {
        std::cout << "diagnostic=" << message << '\n';
        if (expected <= allowance || message.find("CUDA RHF provider exceeds retained allocation allowance") != 0)
          throw std::runtime_error("incorrect provider rejection");
        for (const auto& field : {"free_before_bytes=" + std::to_string(before),
                                  "free_after_bytes=" + std::to_string(sample_after),
                                  "observed_delta_bytes=" + std::to_string(expected),
                                  "allowance_bytes=" + std::to_string(allowance)})
          if (message.find(field) == std::string::npos) throw std::runtime_error("device diagnostic mismatch");
      } else if (expected > allowance || returned != expected) {
        throw std::runtime_error("incorrect provider acceptance");
      }
    }
    if (cleanup_failures) throw std::runtime_error("vendor cleanup failed");
    std::cout << "ENGINEERING_GUARD_PASS; SCIENCE_NOT_RUN\n";
    return 0;
  } catch (const std::exception& error) {
    std::cerr << "ENGINEERING_GUARD_FAIL " << error.what() << '\n';
    return 1;
  }
}
