#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-current-defaults-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/derivative-census-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
"$python" .artifacts/check-source.py > "$output/source-check.txt"
sha256sum .artifacts/derivative-census .artifacts/derivative-census.cpp .artifacts/census-derivative-n1.sh .artifacts/input-{48,96}-{0,1}.txt > "$output/driver.sha256"
export EXPECTED_SOURCE_IDENTITY="$(cat .artifacts/build/source-identity.txt)"
for atoms in 48 96; do
 timeout 800 .artifacts/derivative-census .artifacts/input-$atoms-0.txt .artifacts/input-$atoms-1.txt > "$output/work-$atoms.jsonl" 2> "$output/work-$atoms.log"
done
