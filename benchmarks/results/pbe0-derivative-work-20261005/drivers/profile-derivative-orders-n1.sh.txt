#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-current-defaults-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/derivative-orders-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
"$python" .artifacts/check-source.py > "$output/source-check.txt"
sha256sum .artifacts/derivative-orders .artifacts/derivative-orders.cpp .artifacts/profile-derivative-orders-n1.sh .artifacts/input-{48,96}-{0,1}.txt > "$output/driver.sha256"
export EXPECTED_SOURCE_IDENTITY="$(cat .artifacts/build/source-identity.txt)"
nsys --version > "$output/nsys-version.txt"
for atoms in 48 96; do
 timeout 1000 nsys profile --sample=none --trace=cuda --capture-range=cudaProfilerApi --capture-range-end=stop --force-overwrite=true --output="$output/orders-$atoms" .artifacts/derivative-orders .artifacts/input-$atoms-0.txt .artifacts/input-$atoms-1.txt > "$output/work-$atoms.txt" 2> "$output/profile-$atoms.log"
 nsys stats --report cuda_gpu_kern_sum,cuda_api_sum --format csv --output "$output/statistics-$atoms" "$output/orders-$atoms.nsys-rep" > "$output/stats-$atoms.log" 2>&1
done
