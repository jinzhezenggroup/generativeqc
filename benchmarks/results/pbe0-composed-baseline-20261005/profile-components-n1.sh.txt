#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-composed-baseline-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/profile-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
"$python" .artifacts/check-source.py | tee "$output/source-check.txt"
sha256sum .artifacts/profile-components.py .artifacts/profile-components-n1.sh .artifacts/gpu-env-n1.sh > "$output/inputs.sha256"
for atoms in 48 96; do
 "$python" .artifacts/profile-components.py --atoms "$atoms" --reference .artifacts/endpoint-5768/reference-$atoms.json --output "$output/components-$atoms.json" > "$output/components-$atoms.log" 2>&1
 printf 'profile complete atoms=%s\n' "$atoms"
done
