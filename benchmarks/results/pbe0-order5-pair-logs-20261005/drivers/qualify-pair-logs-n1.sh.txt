#!/usr/bin/env bash
set -euo pipefail
cd /data/jzzeng/qc-1895-order5-pair-logs-20261005
source .artifacts/gpu-env-n1.sh
output="$PWD/.artifacts/gpu-${SLURM_JOB_ID}"
mkdir -p "$output"
trap 'printf "%s\n" "$?" > "$output/job.exit"' EXIT
scontrol show job "$SLURM_JOB_ID" -o > "$output/job.txt"
sha256sum -c .artifacts/build/binaries.sha256 > "$output/binary-check.txt"
sha256sum -c .artifacts/probes/binaries.sha256 > "$output/probe-check.txt"
"$python" .artifacts/check-source.py > "$output/source-check.txt"
sha256sum .artifacts/qualify-pair-logs-n1.sh .artifacts/gpu-env-n1.sh .artifacts/build-probes.sh > "$output/driver.sha256"
export GENERATIVEQC_BECKE_PHASED_PROBE="$PWD/.artifacts/probes/pair.so"
export GENERATIVEQC_PHASED_OWNER_PROBE="$PWD/.artifacts/probes/owner.so"
"$python" -m pytest -q tests/python/test_becke_phased_cuda.py --basetemp="$output/pytest-pair" > "$output/pair.log" 2>&1
"$python" -m pytest -q tests/python/test_stationary_becke_phased_cuda.py --basetemp="$output/pytest-owner" > "$output/owner.log" 2>&1
for sanitizer in memcheck racecheck initcheck synccheck; do
  timeout 240 compute-sanitizer --tool "$sanitizer" --error-exitcode 99 "$python" -m pytest -q tests/python/test_stationary_becke_phased_cuda.py -k 'True and full and False and 96' --basetemp="$output/pytest-$sanitizer" > "$output/$sanitizer.log" 2>&1
done
